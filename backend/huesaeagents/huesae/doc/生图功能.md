```python
    thread_id: str | None
```

## 四、生图节点设计

### 4.1 步骤常量

```python
class ImageStep:
    INPUT = "input"             # 收到用户生图请求
    TAG_GENERATE = "tag_generate"   # 生成Danbooru标签
    TAG_CONFIRM = "tag_confirm"     # 等待用户确认标签
    IMAGE_GENERATE = "image_generate" # 生成图片
    FINISH = "finish"           # 完成
```

### 4.2 节点函数（LangGraph node）

```python
# agents/subagents/image_agent.py

def image_agent_node(state: HuesaeState) -> dict:
    """
    生图节点 - LangGraph 工作流节点

    作为 LangGraph 图的一个节点，接收统一状态，返回部分更新。
    多轮交互由 LangGraph checkpoint 天然支持。
    """
    step = state.get("image_step", ImageStep.INPUT)

    if step == ImageStep.INPUT:
        return _handle_input(state)

    elif step == ImageStep.TAG_CONFIRM:
        return _handle_confirm(state)

    elif step == ImageStep.IMAGE_GENERATE:
        return _handle_generate(state)

    return {"image_step": ImageStep.FINISH}
```

### 4.3 子函数实现

```python
def _handle_input(state: HuesaeState) -> dict:
    """处理用户输入：生成Danbooru标签"""
    user_msg = state["messages"][-1].content
    tags = _generate_tags(user_msg, state["character_id"])
    return {
        "danbooru_tags": tags,
        "image_step": ImageStep.TAG_CONFIRM,
        "messages": [AIMessage(content=f"为你生成了以下标签：\n{', '.join(tags)}\n\n是否使用这些标签生成图片？(是/重新生成)")]
    }

def _handle_confirm(state: HuesaeState) -> dict:
    """处理用户确认"""
    user_msg = state["messages"][-1].content.lower()

    if any(kw in user_msg for kw in ["是", "确认", "确定", "ok", "yes", "可以", "好"]):
        return {"image_step": ImageStep.IMAGE_GENERATE}
    else:
        # 重新生成标签
        original_input = state["messages"][-3].content  # 回溯原始输入
        tags = _generate_tags(original_input, state["character_id"])
        return {
            "danbooru_tags": tags,
            "messages": [AIMessage(content=f"重新生成了标签：\n{', '.join(tags)}\n\n是否使用这些标签？")]
        }

def _handle_generate(state: HuesaeState) -> dict:
    """调用Provider生成图片"""
    tags = state.get("danbooru_tags", [])
    provider_name = state.get("image_provider", "doubao")

    # 调用Provider
    result = _generate_with_provider(tags, provider_name)

    return {
        "generated_image_url": result["url"],
        "image_step": ImageStep.FINISH,
        "messages": [AIMessage(content=f"图片生成完成！\n{result['url']}")]
    }
```

## 五、Provider 设计（开闭原则）

### 5.1 抽象接口

```python
# agents/subagents/image/providers/base.py
from abc import ABC, abstractmethod
from dataclasses import dataclass

@dataclass
class GenerationResult:
    url: str
    provider: str
    prompt: str
    size: str | None = None

class ImageProvider(ABC):
    """图片生成提供者抽象基类"""

    @property
    @abstractmethod
    def name(self) -> str:
        """提供者名称，用于注册和选择"""
        pass

    @abstractmethod
    async def generate(
        self,
        prompt: str,
        size: str = "2K",
        **kwargs
    ) -> GenerationResult:
        """生成图片，返回结果"""
        pass

    def supports_size(self, size: str) -> bool:
        """检查是否支持指定尺寸"""
        return True
```

| 步骤 | 用户输入 | Agent响应 |
|------|---------|----------|
| 1 | "画少女"（不足5字） | "请输入您的提示词" |
| 2 | 用户输入完整提示词 | 询问选择provider |
| 3 | 正常生图流程 | 同上 |

| 2 | "换一版本" 或 "重新生成" | 用相同输入重新生成标签，返回新标签列表 |
| 3 | "可以了" 或 "就这个" | 确认并结束 |

**D. 提示词扩写（自然语言 → 扩写提示词）**

| 步骤 | 用户输入 | Agent响应 |
|------|---------|----------|
| 1 | "扩写提示词：一个银发红瞳的少女" | 返回扩写后的自然语言描述，并额外说明Danbooru标签 |
| 2 | "换一版本" | 重新扩写，返回新版本 |
| 3 | "可以了" | 确认并结束 |

### 9.5 核心设计

#### 9.5.1 状态机设计

```
                    ┌──────────────────────────────────────────────────────┐
                    │                    ImageAgent                        │
                    │                                                    │
  user_input ──────►│  IDLE ──► INTENT_DETECTED ──► WAITING_PROVIDER     │
                    │    │           │                  │                │
                    │    │           ▼                  ▼                │
                    │    │    ┌─────────────────┐  ┌─────────────────┐  │
                    │    │    │ WAITING_PROMPT  │  │ GENERATING      │  │
                    │    │    │ (提示词过短)    │  │ (生成中)         │  │
                    │    │    └─────────────────┘  └─────────────────┘  │
                    │    │                                        │      │
                    │    │      ┌─────────────────────────────────┘      │
                    │    │      ▼                                          │
                    │    │  WAITING_CONFIRM ◄─────────────────────────────┘
                    │    │  (等待确认: 换一张/可以了 或 换版本/可以了)

                    │    └───────────┴───────────────────────────────────┘

                    └─────────────────────────┼───────────────────────────┘
```

**状态说明**：

| 状态 | 说明 | 等待用户输入 |
|------|------|-------------|
| `IDLE` | 初始状态，等待新请求 | 新请求 |
| `WAITING_PROMPT` | 提示词过短，等待用户输入完整提示词 | 完整提示词 |
| `WAITING_PROVIDER` | 等待选择provider | "豆包"/"即梦"/"1"/"2" |
| `GENERATING` | 正在生成图片/标签/扩写 | - |
| `WAITING_CONFIRM` | 等待用户确认结果 | "换一张"/"换一版本"/"可以了" |

### 9.7 接口设计

#### 9.7.1 image_agent.py - 主入口

```python
# agents/subagents/image_agent.py
"""阶段一：独立生图Agent"""

from dataclasses import dataclass, field
from enum import Enum
from typing import Optional
import json

try:
    from ..tools.image import generate_image_by_doubao, generate_image_by_jimeng
except ImportError:
    from huesaeagents.huesae.tools.image import generate_image_by_doubao, generate_image_by_jimeng

from .image_llm import ImageLLM
from .image_prompt import PromptExtractor


class ImageIntent(Enum):
    IMAGE = "image"      # 生图
    TAGS = "tags"        # Danbooru标签
    EXPAND = "expand"    # 扩写
    UNKNOWN = "unknown"


class ImageState(Enum):
    IDLE = "IDLE"
    WAITING_PROMPT = "WAITING_PROMPT"
    WAITING_PROVIDER = "WAITING_PROVIDER"
    GENERATING = "GENERATING"
    WAITING_CONFIRM = "WAITING_CONFIRM"


@dataclass
class ImageContext:
    """生图上下文"""
    intent: ImageIntent = ImageIntent.UNKNOWN
    state: ImageState = ImageState.IDLE
    original_prompt: str = ""
    current_prompt: str = ""
    provider: str = ""
    last_result: Optional[dict] = None
    tag_version: int = 0
    expand_version: int = 0
    # 内部记录用
    _last_generate_prompt: str = ""  # 最近一次生图的prompt（用于换一张）


class ImageAgent:
    """独立生图Agent"""

    # 关键词识别
    IMAGE_KEYWORDS = ["画", "生成图片", "生成图", "画个", "画一张", "生图"]
    TAG_KEYWORDS = ["danbooru", "标签", "tags", "生成标签"]
    EXPAND_KEYWORDS = ["扩写", "展开", "expand", "扩展"]

    # 确认关键词
    CONFIRM_KEYWORDS = ["可以", "好", "了", "yes", "ok", "确定", "确认"]
    REGENERATE_KEYWORDS_IMAGE = ["换一张", "换一张图", "再来", "重新生成"]
    REGENERATE_KEYWORDS_OTHER = ["换一版本", "换版本", "重新生成", "再来一个"]

    def __init__(self, llm_model: str = "doubao"):
        self.llm = ImageLLM(llm_model)
        self.prompt_extractor = PromptExtractor()

    async def process(self, user_input: str, context: ImageContext) -> tuple[str, ImageContext]:
        """
        处理用户输入

        Args:
            user_input: 用户输入
            context: 当前上下文（会被更新）

        Returns:
            (agent_response, updated_context)
        """
        user_input = user_input.strip()

        # 1. 根据状态分发
        if context.state == ImageState.WAITING_PROMPT:
            return self._handle_waiting_prompt(user_input, context)

        elif context.state == ImageState.WAITING_PROVIDER:
            return self._handle_waiting_provider(user_input, context)

        elif context.state == ImageState.WAITING_CONFIRM:
            return self._handle_waiting_confirm(user_input, context)

        elif context.state == ImageState.GENERATING:
            # 生成中忽略输入（理论上不会到达这里）
            return "图片生成中，请稍候...", context

        # 2. IDLE状态：解析新意图
        return self._handle_new_intent(user_input, context)

    # ============== 状态处理器 ==============

    def _handle_new_intent(self, user_input: str, context: ImageContext) -> tuple[str, ImageContext]:
        """处理新意图识别"""
        intent = self._classify_intent(user_input)
        context.intent = intent
        context.original_prompt = user_input

        if intent == ImageIntent.IMAGE:
            return self._start_image_task(user_input, context)
        elif intent == ImageIntent.TAGS:
            return self._start_tags_task(user_input, context)
        elif intent == ImageIntent.EXPAND:
            return self._start_expand_task(user_input, context)
        else:
            return "抱歉，我无法理解您的请求。请说"画一个..."来生成图片，或"生成Danbooru标签"来获取标签。", context

    def _handle_waiting_prompt(self, user_input: str, context: ImageContext) -> tuple[str, ImageContext]:
        """处理等待提示词输入"""
        if len(user_input) < 5:
            return "请输入更详细的提示词（至少5个字）", context

        context.current_prompt = user_input
        context.state = ImageState.WAITING_PROVIDER
        return f"请选择生图方式：1. 豆包  2. 即梦AI", context

    def _handle_waiting_provider(self, user_input: str, context: ImageContext) -> tuple[str, ImageContext]:
        """处理等待provider选择"""
        choice = user_input.lower()
        if choice in ["1", "豆包", "doubao"]:
            context.provider = "doubao"
        elif choice in ["2", "即梦", "jimeng", "即梦ai"]:
            context.provider = "jimeng"
        else:
            return "请选择：1. 豆包  2. 即梦AI", context

        # 开始生成
        return await self._generate_image(context)

    def _handle_waiting_confirm(self, user_input: str, context: ImageContext) -> tuple[str, ImageContext]:
        """处理等待确认"""
        user_lower = user_input.lower()

        # 检查是否确认
        is_confirm = any(kw in user_lower for kw in self.CONFIRM_KEYWORDS)

        if context.intent == ImageIntent.IMAGE:
            # 生图确认
            is_regenerate = any(kw in user_lower for kw in self.REGENERATE_KEYWORDS_IMAGE)
            if is_confirm:
                context.state = ImageState.IDLE
                context.intent = ImageIntent.UNKNOWN
                return "好的，图片已确认！如需其他帮助请随时说～", context
            elif is_regenerate:
                return await self._regenerate_image(context)

        else:
            # 标签/扩写确认
            is_regenerate = any(kw in user_lower for kw in self.REGENERATE_KEYWORDS_OTHER)
            if is_confirm:
                context.state = ImageState.IDLE
                context.intent = ImageIntent.UNKNOWN
                return "好的，已确认！如需其他帮助请随时说～", context
            elif is_regenerate:
                if context.intent == ImageIntent.TAGS:
                    return await self._regenerate_tags(context)
                elif context.intent == ImageIntent.EXPAND:
                    return await self._regenerate_expand(context)

        return "请说"可以了"确认，或"换一张"/"换一版本"重新生成", context

    # ============== 任务启动 ==============

    def _start_image_task(self, prompt: str, context: ImageContext) -> tuple[str, ImageContext]:
        """启动生图任务"""
        # 提取有效提示词
        extracted = self.prompt_extractor.extract(prompt)
        context.current_prompt = extracted

        if len(extracted) < 5:
            context.state = ImageState.WAITING_PROMPT
            return "请输入您的提示词（至少5个字）", context

        context.state = ImageState.WAITING_PROVIDER
        return "请选择生图方式：1. 豆包  2. 即梦AI", context

    def _start_tags_task(self, prompt: str, context: ImageContext) -> tuple[str, ImageContext]:
        """启动标签生成任务"""
        extracted = self.prompt_extractor.extract_for_tags(prompt)
        context.current_prompt = extracted
        context.state = ImageState.GENERATING

        tags = await self.llm.generate_tags(extracted)
        context.last_result = {"tags": tags}
        context.tag_version = 1
        context.state = ImageState.WAITING_CONFIRM

        tags_str = "\n".join(f"- {t}" for t in tags)
        return f"""Danbooru标签：
{tags_str}

是否接受？（"可以了"或"换一版本"）""", context

    def _start_expand_task(self, prompt: str, context: ImageContext) -> tuple[str, ImageContext]:
        """启动提示词扩写任务"""
        extracted = self.prompt_extractor.extract_for_expand(prompt)
        context.current_prompt = extracted
        context.state = ImageState.GENERATING

        expanded, tags = await self.llm.expand_prompt(extracted)
        context.last_result = {"expanded": expanded, "tags": tags}
        context.expand_version = 1
        context.state = ImageState.WAITING_CONFIRM

        tags_str = ", ".join(tags)
        return f"""扩写后的提示词：
{expanded}

（Danbooru标签：{tags_str}）

是否接受？（"可以了"或"换一版本"）""", context

    # ============== 生成逻辑 ==============

    async def _generate_image(self, context: ImageContext) -> tuple[str, ImageContext]:
        """生成图片"""
        context.state = ImageState.GENERATING
        context._last_generate_prompt = context.current_prompt

        try:
            if context.provider == "doubao":
                url = await generate_image_by_doubao(context.current_prompt)
            else:
                url = await generate_image_by_jimeng(context.current_prompt)

            context.last_result = {"url": url}
            context.state = ImageState.WAITING_CONFIRM
            return f"""图片已生成！
{url}

说"换一张"重新生成，或"可以了"结束""", context

        except Exception as e:
            context.state = ImageState.WAITING_CONFIRM
            return f"图片生成失败：{str(e)}\n说"换一张"重新生成，或"可以了"结束", context

    async def _regenerate_image(self, context: ImageContext) -> tuple[str, ImageContext]:
        """重新生成图片（使用相同prompt）"""
        if not context._last_generate_prompt:
            return "没有找到之前的提示词，请重新开始", context

        context.current_prompt = context._last_generate_prompt
        return await self._generate_image(context)

    async def _regenerate_tags(self, context: ImageContext) -> tuple[str, ImageContext]:
        """重新生成标签"""
        context.state = ImageState.GENERATING
        context.tag_version += 1

        tags = await self.llm.generate_tags(context.current_prompt)
        context.last_result = {"tags": tags}
        context.state = ImageState.WAITING_CONFIRM

        tags_str = "\n".join(f"- {t}" for t in tags)
        return f"""Danbooru标签（版本{context.tag_version}）：
{tags_str}

是否接受？""", context

    async def _regenerate_expand(self, context: ImageContext) -> tuple[str, ImageContext]:
        """重新扩写提示词"""
        context.state = ImageState.GENERATING
        context.expand_version += 1

        expanded, tags = await self.llm.expand_prompt(context.current_prompt)
        context.last_result = {"expanded": expanded, "tags": tags}
        context.state = ImageState.WAITING_CONFIRM

        tags_str = ", ".join(tags)
        return f"""扩写后的提示词（版本{context.expand_version}）：
{expanded}

（Danbooru标签：{tags_str}）

是否接受？""", context

    # ============== 工具方法 ==============

    def _classify_intent(self, user_input: str) -> ImageIntent:
        """意图分类"""
        user_lower = user_input.lower()

        if any(kw in user_lower for kw in self.TAG_KEYWORDS):
            return ImageIntent.TAGS
        elif any(kw in user_lower for kw in self.EXPAND_KEYWORDS):
            return ImageIntent.EXPAND
        elif any(kw in user_input for kw in self.IMAGE_KEYWORDS):
            return ImageIntent.IMAGE
        else:
            return ImageIntent.UNKNOWN
```

```python
"""LLM调用 - 标签生成、提示词扩写"""

import json
from langchain.chat_models import init_chat_model


class ImageLLM:
    """生图相关LLM调用"""

    TAG_SYSTEM_PROMPT = """你是一个专业的Danbooru标签生成器。
请将用户描述转换为Danbooru风格的动漫图片标签。

规则：
- 使用下划线连接多词（如 silver_hair）
- 使用复数形式（如 eyes 不是 eye）
- 包含风格标签如 anime, solo 等
- 优先使用常见的Danbooru标签名
- 数量控制在6-12个

直接返回JSON，不要有多余文字：
{"tags": ["tag1", "tag2", ...]}"""

    EXPAND_SYSTEM_PROMPT = """你是一个专业的AI生图提示词扩写专家。
请将用户的简短描述扩写成详细、生动的AI生图提示词。

规则：
- 保持原意但增加细节描述
- 添加氛围、光照、视角等描述
- 使用适合AI生图的专业术语
- 扩写后的描述应控制在100-200字

同时返回对应的Danbooru标签。

直接返回JSON，不要有多余文字：
{"expanded": "扩写后的详细描述", "tags": ["tag1", "tag2", ...]}"""

    def __init__(self, model_name: str = "doubao"):
        self.llm = init_chat_model(model_name, temperature=0.8)

    async def generate_tags(self, description: str) -> list[str]:
        """生成Danbooru标签"""
        response = await self.llm.ainvoke(
            [("system", self.TAG_SYSTEM_PROMPT), ("human", description)]
        )
        try:
            result = json.loads(response.content)
            return result.get("tags", [])
        except:
            # 降级处理：简单分词
            return [w.strip() for w in description.split() if len(w) > 2]

    async def expand_prompt(self, description: str) -> tuple[str, list[str]]:
        """扩写提示词"""
        response = await self.llm.ainvoke(
            [("system", self.EXPAND_SYSTEM_PROMPT), ("human", description)]
        )
        try:
            result = json.loads(response.content)
            return result.get("expanded", description), result.get("tags", [])
        except:
            return description, []
```

```python
# agents/subagents/image_prompt.py
"""提示词提取工具"""


class PromptExtractor:
    """提示词提取器"""

    # 生图前缀
    IMAGE_PREFIXES = [
        "画", "生成图片", "生成图", "画个", "画一张",
        "画一幅", "生成一幅", "帮我画", "给我画",
        "生图", "生成"
    ]

    # 标签任务前缀
    TAG_PREFIXES = [
        "danbooru", "标签", "tags", "生成标签",
        "生成danbooru", "给我标签", "转标签"
    ]

    # 扩写任务前缀
    EXPAND_PREFIXES = [
        "扩写", "展开", "expand", "扩展", "详细化",
        "扩写提示词", "展开提示词"
    ]

    def extract(self, user_input: str) -> str:
        """从用户输入中提取生图提示词"""
        text = user_input.strip()

        # 去掉生图前缀
        for prefix in self.IMAGE_PREFIXES:
            if text.startswith(prefix):
                text = text[len(prefix):].strip()
                break

        # 去掉"一个"/"一只"等开头的量词
        for quantifier in ["一个", "一只", "一幅", "一张"]:
            if text.startswith(quantifier):
                text = text[len(quantifier):].strip()
                break

        return text

    def extract_for_tags(self, user_input: str) -> str:
        """提取标签生成任务的描述"""
        text = user_input.strip()

        # 去掉标签前缀
        for prefix in self.TAG_PREFIXES:
            if prefix in text.lower():
                text = text.lower().replace(prefix, "").strip()
                break

        # 去掉"的"或"是"开头
        if text.startswith("的") or text.startswith("是"):
            text = text[1:].strip()

        return text

    def extract_for_expand(self, user_input: str) -> str:
        """提取扩写任务的描述"""
        text = user_input.strip()

        # 去掉扩写前缀
        for prefix in self.EXPAND_PREFIXES:
            if prefix in text:
                text = text.replace(prefix, "").strip()
                break

        # 去掉冒号后的内容（如果有）
        if ":" in text:
            text = text.split(":", 1)[1].strip()

        return text
```

```python
# agents/subagents/image_agent.py
"""阶段一：独立生图Agent（快速跑通）"""

from enum import Enum
from langchain.chat_models import init_chat_model
from .image_providers import DoubaoProvider, JimengProvider

class ImageTask(str, Enum):
    """生图任务类型"""
    GENERATE_IMAGE = "generate_image"      # 生图
    GENERATE_TAGS = "generate_tags"       # Danbooru标签
    EXPAND_PROMPT = "expand_prompt"       # 扩写提示词

class ImageAgent:
    """独立生图Agent"""

    def __init__(self, model_name: str = "doubao"):
        self.llm = init_chat_model(model_name)
        self.doubao = DoubaoProvider()
        self.jimeng = JimengProvider()

    async def run(self, user_input: str, history: list) -> str:
        """主入口"""
        task = self._classify_task(user_input)

        if task == ImageTask.GENERATE_IMAGE:
            return await self._handle_image_generation(user_input, history)
        elif task == ImageTask.GENERATE_TAGS:
            return await self._handle_tag_generation(user_input)
        elif task == ImageTask.EXPAND_PROMPT:
            return await self._handle_prompt_expansion(user_input)
        else:
            return "抱歉，无法理解您的请求"

    def _classify_task(self, user_input: str) -> ImageTask:
        """意图分类"""
        user_lower = user_input.lower()

        if any(kw in user_lower for kw in ["danbooru", "标签", "tags"]):
            return ImageTask.GENERATE_TAGS
        elif any(kw in user_lower for kw in ["扩写", "展开", "expand"]):
            return ImageTask.EXPAND_PROMPT
        else:
            return ImageTask.GENERATE_IMAGE

    async def _handle_image_generation(self, user_input: str, history: list) -> str:
        """处理生图请求"""
        # 提取提示词（去掉"画"、"生成图片"等前缀）
        prompt = self._extract_prompt(user_input)

        # 检查提示词长度
        if len(prompt) < 5:
            return "请输入您的提示词（至少5个字）"

        # 询问provider选择
        return "请选择生图方式：1. 豆包  2. 即梦AI"

    async def _handle_provider_selection(self, provider_choice: str, prompt: str, history: list) -> str:
        """处理provider选择和图片生成"""
        if provider_choice in ["1", "豆包"]:
            result = await self.doubao.generate(prompt)
        elif provider_choice in ["2", "即梦", "即梦AI"]:
            result = await self.jimeng.generate(prompt)
        else:
            return "请选择：1. 豆包  2. 即梦AI"

        history.append({"role": "assistant", "content": f"[生成中...] 图片已生成！{result.url}"})

        # 追加换图/确认提示
        history.append({
            "role": "assistant",
            "content": f"图片已生成！{result.url}\n说"换一张"重新生成，或"可以了"结束"
        })
        return history[-1]["content"]

    async def _handle_regenerate(self, history: list) -> str:
        """处理换一张"""
        # 找到最近一次生图的prompt
        for msg in reversed(history):
            if "图片已生成" in msg.get("content", ""):
                continue
            # 从更早的消息中提取prompt...
        # 简化：重新调用doubao
        return "正在重新生成..."

    async def _handle_tag_generation(self, user_input: str) -> str:
        """处理Danbooru标签生成"""
        # 提取任务描述
        prompt = user_input
        for kw in ["danbooru", "标签", "tags", "生成"]:
            prompt = prompt.replace(kw, "").strip()

        if not prompt:
            return "请输入要生成标签的描述"

        tags = await self._generate_danbooru_tags(prompt)
        tags_str = "\n".join(f"- {t}" for t in tags)

        return f"""Danbooru标签：
{tags_str}

是否接受？（"可以了"或"换一版本"）"""

    async def _handle_tag_confirm(self, user_input: str, tags_prompt: str, history: list) -> str:
        """处理标签确认"""
        user_lower = user_input.lower()

        if any(kw in user_lower for kw in ["可以", "好", "了", "yes", "ok"]):
            return "好的，标签已确认！"
        elif any(kw in user_lower for kw in ["换", "重新", "版本"]):
            tags = await self._generate_danbooru_tags(tags_prompt)
            tags_str = "\n".join(f"- {t}" for t in tags)
            return f"""Danbooru标签（新版）：
{tags_str}

是否接受？"""
        else:
            return "请说"可以了"确认，或"换一版本"重新生成"

    async def _handle_prompt_expansion(self, user_input: str) -> str:
        """处理提示词扩写"""
        # 提取要扩写的描述
        prompt = user_input
        for kw in ["扩写", "展开", "expand"]:
            prompt = prompt.replace(kw, "").strip()

        if not prompt:
            return "请输入要扩写的提示词"

        result = await self._expand_prompt(prompt)
        danbooru_tags = await self._generate_danbooru_tags(prompt)
        tags_str = ", ".join(danbooru_tags)

        return f"""扩写后的提示词：
{result.expanded}

（Danbooru标签：{tags_str}）

是否接受？（"可以了"或"换一版本"）"""

    async def _expand_prompt(self, prompt: str) -> dict:
        """LLM扩写提示词"""
        expand_prompt = f"""请扩写以下提示词，使其更加详细具体，适合用于AI生图：

原始提示词：{prompt}

请返回JSON格式：
{{"expanded": "扩写后的详细描述"}}
"""
        response = await self.llm.ainvoke(expand_prompt)
        import json
        return json.loads(response.content)

    async def _generate_danbooru_tags(self, description: str) -> list[str]:
        """LLM生成Danbooru标签"""
        tag_prompt = f"""将以下描述转换为Danbooru风格的标签（动漫风格图片标签）：

描述：{description}

标签要求：
- 使用下划线连接多词（如 silver_hair）
- 使用复数形式（如 eyes 不是 eye）
- 包含风格标签如 anime, solo 等
- 数量控制在5-10个

返回JSON格式：
{{"tags": ["tag1", "tag2", ...]}}
"""
        response = await self.llm.ainvoke(tag_prompt)
        import json
        return json.loads(response.content)["tags"]

    def _extract_prompt(self, user_input: str) -> str:
        """提取生图提示词"""
        # 去掉常见前缀
        for prefix in ["画", "生成图片", "生成图", "画一个", "画一张"]:
            if user_input.startswith(prefix):
                return user_input[len(prefix):].strip()
        return user_input
```
